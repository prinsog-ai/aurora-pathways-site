Solidity 
// TaskEscrowV2.sol — createJob function
function createJob(
    address payee,
    string calldata description,
    uint256 deadline
) external returns (uint256 jobId) {
    require(payee != address(0), "Invalid payee");
    require(msg.value > 0, "Must fund job");
    require(deadline > block.timestamp, "Invalid deadline");

    jobId = nextJobId++;
    jobs[jobId] = Job({
        client: msg.sender,
        payee: payee,
        amount: msg.value,
        deadline: deadline,
        released: false,
        description: description
    });
    emit JobCreated(jobId, msg.sender, payee, msg.value);
}

// Connect & call createJob from your dApp
import { BrowserProvider, Contract, parseEther } from 'ethers';

const provider = new BrowserProvider(window.ethereum);
const signer = await provider.getSigner();

const escrow = new Contract(
  '0x34623Df95A9F2f48c056BE79d2A65C5B4DE2B30e',
  TASK_ESCROW_ABI,
  signer
);

const tx = await escrow.createJob(
  payeeAddress,
  'Build landing page',
  Math.floor(Date.now() / 1000) + 86400 * 7,
  { value: parseEther('1.5') }
);
console.log('Job created: ', tx.hash);
